function displayPage(page) {
      currentPage = page;
      const totalProducts = filteredProductsGlobal.length;
      if (totalProducts === 0) {
        renderProducts([]);
        document.getElementById('results-counter').innerText = `Mostrando 0 artículos`;
        return;
      }
      const startIndex = (currentPage - 1) * productsPerPage;
      const endIndex = Math.min(startIndex + productsPerPage, totalProducts);
      renderProducts(filteredProductsGlobal.slice(startIndex, endIndex));
      document.getElementById('results-counter').innerText = `Mostrando ${startIndex + 1}-${endIndex} de ${totalProducts} artículos`;
    }
