async function loadCatalogData() {
      let data = null;
      try {
        const response = await fetch('docs/data/catalog.json');
        if (response.ok) {
          data = await response.json();
        } else {
          const responseFallback = await fetch('data/catalog.json');
          if (responseFallback.ok) {
            data = await responseFallback.json();
          }
        }
      } catch (e) {
        console.warn("Error cargando catálogo remoto, intentando fallback local...");
        try {
          const responseFallback = await fetch('data/catalog.json');
          if (responseFallback.ok) {
            data = await responseFallback.json();
          }
        } catch (inner_e) {
          console.warn("Fallo total de red. Utilizando catálogo de fallbacks de alta gama.");
        }
      }

      if (data) {
        let productsArray = [];
        if (Array.isArray(data)) {
          productsArray = data;
        } else if (data && Array.isArray(data.products)) {
          productsArray = data.products;
          // Guardar datos en el motor universal si es el formato del crawler
          window.VIA_MX_PRO_ENGINE.catalog_data = data;
        }

        if (productsArray.length > 0) {
          productsArray.forEach(jsonProd => {
            // Normalizar producto del crawler nuevo al formato del chasis base
            const id = jsonProd.id || jsonProd.product_id;
            const plat = jsonProd.platform || jsonProd.platform_source || (id && id.includes("amazon") ? "amazon_mx" : "mercadolibre_mx");
            const clean_key = id ? id.replace("card_amazon_mx_", "").replace("card_mercadolibre_mx_", "").replace("card_", "") : "";
            
            // Determinar origen
            let source = jsonProd.source;
            if (!source) {
              if (plat === "amazon_mx") source = "Amazon México";
              else if (plat === "mercadolibre_mx") source = "Mercado Libre";
              else if (plat === "sears_mx") source = "SEARS";
              else if (plat === "palacio_hierro") source = "Palacio de Hierro";
              else if (plat === "aliexpress") source = "AliExpress";
              else source = "VíaMX Pro";

            let price = jsonProd.price;
            let oldPrice = jsonProd.oldPrice;
            if (jsonProd.pricing) {
              price = jsonProd.pricing.current_price;
              oldPrice = jsonProd.pricing.original_price;

            let basePath = jsonProd.basePath || jsonProd.clean_affiliate_link;

              }
            }

            let metric = jsonProd.metric || "Calibración verificada por expertos";
            let altTitle = jsonProd.alternative_title || "Producto alternativo oficial";
            let altPrice = jsonProd.alternative_price || (price ? price * 1.2 : 1000);
            let altLink = jsonProd.alternative_link || basePath;
