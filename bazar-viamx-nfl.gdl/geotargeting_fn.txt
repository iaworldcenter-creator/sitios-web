function applyClientGeotargeting() {
      const urlParams = new URLSearchParams(window.location.search);
      const simCountry = urlParams.get('sim_country');

      if (simCountry) {
        detectedCountry = simCountry.toUpperCase();
        console.log(`Geotargeting Simulado: País forzado a [${detectedCountry}] por URL.`);
      } else {
        try {
          const controller = new AbortController();
          const timeoutId = setTimeout(() => controller.abort(), 600);
          const res = await fetch('https://ipapi.co/json/', { signal: controller.signal });
          clearTimeout(timeoutId);
          if (res.ok) {
            const data = await res.json();
            if (data.country_code) {
              detectedCountry = data.country_code.toUpperCase();
              console.log(`Geotargeting Real: IP geolocalizada en [${detectedCountry}]`);
            }
          }
        } catch (e) {
          console.warn("Geotargeting: Error o timeout al geolocalizar IP. Fallback a MX.");
        }
      }
